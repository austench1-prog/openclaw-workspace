# Camarilla Weekly Formula - TOS ThinkScript
# Version: v2
# Fix: Lines only display within the target week, no extension to following weeks
#
# WH  = Last week High
# WL  = Last week Low
# WC  = Last Friday close
# WO  = Last Monday open
# P   = (WH + WL + WC + WO) / 4
# B   = 2 * P
# NWH = B - WL    <- Next Week High
# NWL = B - WH    <- Next Week Low

declare upper;

# Last completed week's data
def WH = high(period = AggregationPeriod.WEEK)[1];
def WL = low(period = AggregationPeriod.WEEK)[1];
def WC = close(period = AggregationPeriod.WEEK)[1];
def WO = open(period = AggregationPeriod.WEEK)[1];

# Formula
def P   = (WH + WL + WC + WO) / 4;
def B   = 2 * P;
def NWH = B - WL;
def NWL = B - WH;
def MID = (NWH + NWL) / 2;

# v2 Fix: Only show lines within the CURRENT week (week[0])
# Lines will NOT extend beyond this week's bars
def thisWeekBar = GetYYYYMMDD() >= GetYYYYMMDD(period = AggregationPeriod.WEEK)[0];
def nextWeekBar = GetYYYYMMDD() >= GetYYYYMMDD(period = AggregationPeriod.WEEK)[-1];
def showThisWeek = thisWeekBar and !nextWeekBar and !IsNaN(WH) and !IsNaN(WO);

# Plots - only within current week
plot NextWeekHigh = if showThisWeek then NWH else Double.NaN;
NextWeekHigh.SetDefaultColor(CreateColor(0, 180, 255));
NextWeekHigh.SetLineWeight(3);
NextWeekHigh.SetPaintingStrategy(PaintingStrategy.HORIZONTAL);
NextWeekHigh.HideBubble();

plot NextWeekLow = if showThisWeek then NWL else Double.NaN;
NextWeekLow.SetDefaultColor(CreateColor(0, 180, 255));
NextWeekLow.SetLineWeight(3);
NextWeekLow.SetPaintingStrategy(PaintingStrategy.HORIZONTAL);
NextWeekLow.HideBubble();

plot MidLine = if showThisWeek then MID else Double.NaN;
MidLine.SetDefaultColor(Color.YELLOW);
MidLine.SetLineWeight(1);
MidLine.SetPaintingStrategy(PaintingStrategy.HORIZONTAL);
MidLine.HideBubble();

AddCloud(NextWeekHigh, NextWeekLow, CreateColor(0, 180, 255), CreateColor(0, 180, 255));

# Labels
AddLabel(yes, "Weekly Camarilla v2", CreateColor(0, 180, 255));
AddLabel(yes, "WH: " + AsPrice(WH) + "  WL: " + AsPrice(WL), Color.GRAY);
AddLabel(yes, "WO(Mon): " + AsPrice(WO) + "  WC(Fri): " + AsPrice(WC), Color.GRAY);
AddLabel(yes, "NWH: " + AsPrice(NWH), CreateColor(0, 180, 255));
AddLabel(yes, "MID: " + AsPrice(MID), Color.YELLOW);
AddLabel(yes, "NWL: " + AsPrice(NWL), CreateColor(0, 180, 255));

# Alerts
Alert(showThisWeek and close >= NWH, "Touched Next Week High: " + AsPrice(NWH), Alert.BAR, Sound.Bell);
Alert(showThisWeek and close <= NWL, "Touched Next Week Low: " + AsPrice(NWL), Alert.BAR, Sound.Bell);
