# Camarilla Weekly Formula - TOS ThinkScript
# Version: v3
# Fix: WC must be Friday's close (last trading day of the week)
#      WO must be Monday's open (first trading day of the week)
#      Only display after previous week is fully closed
#
# WH  = Last week High (real market data)
# WL  = Last week Low (real market data)
# WC  = Last FRIDAY close (real market data - day close only)
# WO  = Last Monday open (real market data - day open only)
# P   = (WH + WL + WC + WO) / 4
# B   = 2 * P
# NWH = B - WL    <- This Week High prediction
# NWL = B - WH    <- This Week Low prediction

declare upper;

# Last completed week's High and Low (weekly bar - reliable)
def WH = high(period = AggregationPeriod.WEEK)[1];
def WL = low(period = AggregationPeriod.WEEK)[1];

# Friday close: get the close of the last day of the previous week
# Using daily close on Friday (day of week = 6 in TOS = Friday)
# Walk back on daily bars to find Friday's close
def dayOfWeek = GetDayOfWeek(GetYYYYMMDD());
def lastFridayClose = close(period = AggregationPeriod.DAY)[
    if dayOfWeek == 1 then 1   # Monday: go back 1 day (Friday)
    else if dayOfWeek == 2 then 2   # Tuesday: go back 2 days
    else if dayOfWeek == 3 then 3   # Wednesday: go back 3 days
    else if dayOfWeek == 4 then 4   # Thursday: go back 4 days
    else if dayOfWeek == 5 then 5   # Friday: go back 5 days (last Friday)
    else 0
];
def WC = lastFridayClose;

# Monday open: first trading day of the previous week
# Get open from the Monday of the previous week (5 trading days back from this Friday)
def WO = open(period = AggregationPeriod.WEEK)[1];

# Formula
def P   = (WH + WL + WC + WO) / 4;
def B   = 2 * P;
def NWH = B - WL;
def NWL = B - WH;
def MID = (NWH + NWL) / 2;

# Only show when previous week is complete (WC is a real Friday close)
# Do not show if WH or WL is from incomplete week
def show = !IsNaN(WH) and !IsNaN(WO) and !IsNaN(WC) and WC > 0;

# Plots
plot NextWeekHigh = if show then NWH else Double.NaN;
NextWeekHigh.SetDefaultColor(CreateColor(0, 180, 255));
NextWeekHigh.SetLineWeight(3);
NextWeekHigh.SetPaintingStrategy(PaintingStrategy.HORIZONTAL);
NextWeekHigh.HideBubble();

plot NextWeekLow = if show then NWL else Double.NaN;
NextWeekLow.SetDefaultColor(CreateColor(0, 180, 255));
NextWeekLow.SetLineWeight(3);
NextWeekLow.SetPaintingStrategy(PaintingStrategy.HORIZONTAL);
NextWeekLow.HideBubble();

plot MidLine = if show then MID else Double.NaN;
MidLine.SetDefaultColor(Color.YELLOW);
MidLine.SetLineWeight(1);
MidLine.SetPaintingStrategy(PaintingStrategy.HORIZONTAL);
MidLine.HideBubble();

AddCloud(NextWeekHigh, NextWeekLow, CreateColor(0, 180, 255), CreateColor(0, 180, 255));

# Labels
AddLabel(yes, "Weekly Camarilla v3", CreateColor(0, 180, 255));
AddLabel(yes, "WH: " + AsPrice(WH) + "  WL: " + AsPrice(WL), Color.GRAY);
AddLabel(yes, "WO(Mon): " + AsPrice(WO) + "  WC(Fri): " + AsPrice(WC), Color.GRAY);
AddLabel(yes, "NWH: " + AsPrice(NWH), CreateColor(0, 180, 255));
AddLabel(yes, "MID: " + AsPrice(MID), Color.YELLOW);
AddLabel(yes, "NWL: " + AsPrice(NWL), CreateColor(0, 180, 255));

# Alerts
Alert(show and close >= NWH, "Touched Week High: " + AsPrice(NWH), Alert.BAR, Sound.Bell);
Alert(show and close <= NWL, "Touched Week Low: " + AsPrice(NWL), Alert.BAR, Sound.Bell);
